local function detectSea()
    local pid = game.PlaceId
    if pid == 2753915549 then return 1 end
    if pid == 4442272183 then return 2 end
    if pid == 7449423635 then return 3 end

    local map = workspace:FindFirstChild("Map")
    if map then
        if map:FindFirstChild("Turtle") or map:FindFirstChild("PortTown") or map:FindFirstChild("CakeLoaf") then
            return 3
        end
        if map:FindFirstChild("Dressrosa") or map:FindFirstChild("Kingdom of Rose") or map:FindFirstChild("GreenBit") then
            return 2
        end
    end

    local ok, lv = pcall(function() return LP.Data.Level.Value end)
    if ok and tonumber(lv) then
        if lv >= 1500 then return 3 end
        if lv >= 700 then return 2 end
    end
    return 1
end

local function connect(signal, fn)
    local c = signal:Connect(fn)
    table.insert(State.connections, c)
    return c
end

local function safe(fn, ...)
    local ok, err = pcall(fn, ...)
    if not ok then warn("[Vanh Hub] " .. tostring(err)) end
end

local function getChar()
    local char = LP.Character
    if not char then return end
    return char,
        char:FindFirstChildOfClass("Humanoid"),
        char:FindFirstChild("HumanoidRootPart")
end

local function isPointer(input)
    return input.UserInputType == Enum.UserInputType.MouseButton1
        or input.UserInputType == Enum.UserInputType.Touch
end

local function getRemote(name)
    local remotes = Svc.ReplicatedStorage:FindFirstChild("Remotes")
    return remotes and remotes:FindFirstChild(name) or nil
end

local function invokeCommF(...)
    local comm = getRemote("CommF_")
    if not comm then return nil end
    local ok, res = pcall(comm.InvokeServer, comm, ...)
    return ok and res or nil
end

local function playerNumber(name)
    local ok, v = pcall(function() return LP.Data[name].Value end)
    if ok and tonumber(v) then return tonumber(v) end
    local ok2, v2 = pcall(function() return LP.leaderstats[name].Value end)
    return ok2 and tonumber(v2) or nil
end

local function getPlayerRace()
    local ok, r = pcall(function() return LP.Data.Race.Value end)
    return ok and tostring(r) or "Human"
end

local function restoreNoclip()
    for part in pairs(State.noclipped) do
        if part and part.Parent then part.CanCollide = true end
    end
    table.clear(State.noclipped)
end

local function stopFarmBody()
    if State.farmBV then
        pcall(function() State.farmBV:Destroy() end)
        State.farmBV = nil
    end
end

local function stopTpBody()
    if State.tpBV then
        pcall(function() State.tpBV:Destroy() end)
        State.tpBV = nil
    end
end

local function cancelTeleport()
    State.flight = nil
    State.currentTween = nil
    State.isTeleporting = false
    stopTpBody()
    local _, _, root = getChar()
    if root then
        root.AssemblyLinearVelocity = Vector3.zero
        root.AssemblyAngularVelocity = Vector3.zero
    end
end

local UI = {}

function UI.new(class, props, parent)
    local o = Instance.new(class)
    for k, v in pairs(props or {}) do o[k] = v end
    if parent then o.Parent = parent end
    return o
end

function UI.corner(obj, r) return UI.new("UICorner", { CornerRadius = UDim.new(0, r) }, obj) end

function UI.stroke(obj, color, thickness, transparency)
    return UI.new("UIStroke", {
        Color = color, Thickness = thickness or 1, Transparency = transparency or 0,
        ApplyStrokeMode = Enum.ApplyStrokeMode.Border,
    }, obj)
end

function UI.grad(obj, c1, c2, rotation)
    obj.BackgroundColor3 = Color3.new(1, 1, 1)
    return UI.new("UIGradient", { Color = ColorSequence.new(c1, c2), Rotation = rotation or 0 }, obj)
end

function UI.tween(obj, props, t)
    Svc.TweenService:Create(obj,
        TweenInfo.new(t or 0.15, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), props):Play()
end

function UI.text(parent, str, size, color, font, props)
    local l = UI.new("TextLabel", {
        BackgroundTransparency = 1, BorderSizePixel = 0, Text = str, TextSize = size or 13,
        TextColor3 = color or T.text, Font = font or Enum.Font.GothamMedium,
        TextXAlignment = Enum.TextXAlignment.Left, TextTruncate = Enum.TextTruncate.AtEnd,

local function makeDraggable(handle, dragTarget, onTap)
    handle.InputBegan:Connect(function(input)
        if not isPointer(input) then return end
        local startPointer = input.Position
        local startPos = dragTarget.Position
        local moved = false
        beginDrag({
            move = function(i)
                local d = i.Position - startPointer
                if not moved and d.Magnitude > 5 then moved = true end
                if moved then
                    dragTarget.Position = UDim2.new(
                        startPos.X.Scale, startPos.X.Offset + d.X,
                        startPos.Y.Scale, startPos.Y.Offset + d.Y)
                end
            end,
            finish = function()
                if not moved and onTap then onTap() end
            end,
        })
    end)
end

local function slider(page, label, key, min, max, step, cb, unit)
    local row = newRow(page, 54)
    UI.text(row, label, 13, T.text, nil, { Size = UDim2.new(0.65, 0, 0, 24), Position = UDim2.new(0, 12, 0, 4) })
    local valueLbl = UI.text(row, "", 12, T.accent, Enum.Font.GothamBold, {
        Size = UDim2.new(0.35, -12, 0, 24), Position = UDim2.new(0.65, 0, 0, 4),
        TextXAlignment = Enum.TextXAlignment.Right,
    })
    local track = UI.new("Frame", {
        Size = UDim2.new(1, -24, 0, 6), Position = UDim2.new(0, 12, 0, 36),
        BackgroundColor3 = T.off, BorderSizePixel = 0,
    }, row)
    UI.corner(track, 3)
    local fill = UI.new("Frame", { Size = UDim2.new(0, 0, 1, 0), BorderSizePixel = 0 }, track)
    UI.corner(fill, 3)
    UI.grad(fill, T.accent, T.accent2, 0)
    local knob = UI.new("Frame", {
        Size = UDim2.new(0, 14, 0, 14), AnchorPoint = Vector2.new(0.5, 0.5),
        Position = UDim2.new(0, 0, 0.5, 0), BackgroundColor3 = Color3.new(1, 1, 1), BorderSizePixel = 0,
    }, track)
    UI.corner(knob, 7)
    local hit = UI.new("TextButton", {
        Size = UDim2.new(1, 0, 0, 26), Position = UDim2.new(0, 0, 0, 28),
        BackgroundTransparency = 1, Text = "", BorderSizePixel = 0,
    }, row)

    local function render()
        local frac = (config[key] - min) / (max - min)
        fill.Size = UDim2.new(frac, 0, 1, 0)
        knob.Position = UDim2.new(frac, 0, 0.5, 0)
        valueLbl.Text = tostring(config[key]) .. (unit or "")
    end
    local function setFromX(x)
        local w = track.AbsoluteSize.X
        if w <= 0 then return end
        local frac = math.clamp((x - track.AbsolutePosition.X) / w, 0, 1)
        local v = math.clamp(math.floor((min + frac * (max - min)) / step + 0.5) * step, min, max)
        if v ~= config[key] then
            config[key] = v
            render()
            if cb then safe(cb, v) end
        end
    end
    hit.InputBegan:Connect(function(input)
        if not isPointer(input) then return end
        page.ScrollingEnabled = false
        setFromX(input.Position.X)
        beginDrag({
            move = function(i) setFromX(i.Position.X) end,
            finish = function() page.ScrollingEnabled = true end,
        })
    end)
    render()
end

-- =====================================================
--  HỆ THỐNG BAY MỚI (Flight)
-- =====================================================
local FLIGHT_PULL_DIST = 45
local FLIGHT_VERIFY_DIST = 60
local FLIGHT_ARRIVE_DIST = 3
local FLIGHT_SNAP_DIST = 25

local function getSafeFlySpeed()
    local spd = getEffectiveTpSpeed()
    local cap = config.autoTweenSpeed and 400 or 350
    spd = math.clamp(spd, 50, cap)
    if os.clock() < State.slowUntil then
        spd = spd * math.max(0.4, 0.7 ^ State.rbLevel)
    else
        State.rbLevel = 0
    end
    return math.max(spd, 90)
end

local function markRubberBand()
    State.rbCount = State.rbCount + 1
    State.rbLevel = math.min(State.rbLevel + 1, 4)
    State.slowUntil = os.clock() + 30
    if os.clock() - State.lastRbNotify > 6 then
        State.lastRbNotify = os.clock()
        notify("Bị kéo ngược", "Lần " .. State.rbCount .. ": tự giảm tốc độ bay", 3, T.bad)
    end
end

local function recordSet(root)
    State.lastSetRoot = root
    State.lastSetPos = root.Position
    State.lastSetAt = os.clock()
end

local function detectPull(root)
    if State.lastSetRoot == root and State.lastSetPos and os.clock() - State.lastSetAt < 0.5 then
        if (root.Position - State.lastSetPos).Magnitude > FLIGHT_PULL_DIST then
            State.lastSetPos = nil
            markRubberBand()
            return true
        end
    end
    return false
end

local function zeroVelocity(root)
    root.AssemblyLinearVelocity = Vector3.zero
    root.AssemblyAngularVelocity = Vector3.zero
end

local function finishFlight()
    local f = State.flight
    State.flight = nil
    State.isTeleporting = false
    stopTpBody()
    local _, _, r = getChar()
    if r then zeroVelocity(r) end
    State.lastSetPos = nil
    if not config.noclip and not anyAutoActive() then
        restoreNoclip()
    end
    State.busyUntil = math.max(State.busyUntil, os.clock() + ((f and f.holdAfter) or 0.3))
end

local function updateFlight(dt)
    local f = State.flight
    if not f then return end
    local _, hum, root = getChar()
    if not (hum and root) or hum.Health <= 0 then
        cancelTeleport()
        return
    end
    dt = math.clamp(dt or 0.016, 0.001, 0.05)
    local now = os.clock()
    detectPull(root)

    if f.phase == "fly" then
        local diff = f.cf.Position - root.Position
        local dist = diff.Magnitude
        if dist <= FLIGHT_ARRIVE_DIST then
            root.CFrame = f.cf
            zeroVelocity(root)
            f.phase = "settle"
            f.settleUntil = now + 0.8
            State.statusText = "Đã đến nơi, đang ổn định vị trí..."
        else
            local ramp = math.clamp(0.25 + (now - f.startedAt) / 0.4, 0.25, 1)
            local brake = math.clamp(dist / 40, 0.35, 1)
            local spd = math.max(getSafeFlySpeed() * ramp * brake, 40)
            local step = math.min(dist, spd * dt)
            root.CFrame = CFrame.new(root.Position + diff.Unit * step) * f.rot
            zeroVelocity(root)
            State.statusText = "Đang bay tới đích (" .. math.floor(dist) .. " studs)"
        end
    else
        zeroVelocity(root)
        if now >= f.settleUntil then
            local flat = Vector3.new(root.Position.X - f.cf.Position.X, 0, root.Position.Z - f.cf.Position.Z).Magnitude
            if flat > FLIGHT_VERIFY_DIST then
                markRubberBand()
                if f.retries > 0 then
                    f.retries = f.retries - 1
                    f.phase = "fly"
                    f.startedAt = now
                else
                    finishFlight()
                    State.statusText = "Server kéo ngược, đã thử lại nhiều lần"
                    return
                end
            else
                finishFlight()
                return
            end
        end
    end
    recordSet(root)
end

connect(Svc.RunService.Heartbeat, function(dt)
    pcall(updateFlight, dt)
end)

local function teleportTo(cf, retries)
    if typeof(cf) == "Vector3" then
        cf = CFrame.new(cf)
    elseif typeof(cf) == "Instance" and cf:IsA("BasePart") then
        cf = cf.CFrame
    end
    if typeof(cf) ~= "CFrame" then return 0 end
    local char, hum, root = getChar()
    if not (char and hum and root) or hum.Health <= 0 then return 0 end
    cancelTeleport()
    if hum.Sit then hum.Sit = false end

    local dist = (root.Position - cf.Position).Magnitude
    if (not config.smoothTp) and dist <= 100 then
        zeroVelocity(root)
        root.CFrame = cf
        State.lastSetPos = nil
        return 0
    end

    local dur = dist / getSafeFlySpeed() + 1.2
    State.tpToken = State.tpToken + 1
    State.flight = {
        cf = cf,
        rot = cf - cf.Position,
        retries = (retries == nil) and 2 or retries,
        phase = "fly",
        startedAt = os.clock(),
        settleUntil = 0,
    }
    State.isTeleporting = true
    State.lastSetPos = nil
    stopTpBody()
    State.tpBV = UI.new("BodyVelocity", {
        MaxForce = Vector3.new(1e5, 1e5, 1e5),
        Velocity = Vector3.zero,
    }, root)
    return dur
end

local function manualTeleport(cf, retries)
    local dur = teleportTo(cf, retries)
    if State.flight then
        State.flight.holdAfter = 6
    else
        State.busyUntil = os.clock() + 6
    end
    return dur
end

local function moveTowards(root, destPos, lookPos, dt)
    detectPull(root)
    dt = math.clamp(dt or 0.016, 0.001, 0.05)

    local diff = destPos - root.Position
    local dist = diff.Magnitude
    local goal
    if (destPos - lookPos).Magnitude < 0.1 or (math.abs(destPos.X - lookPos.X) < 0.05 and math.abs(destPos.Z - lookPos.Z) < 0.05) then
        goal = CFrame.new(destPos) * CFrame.Angles(math.rad(-90), 0, 0)
    else
        goal = CFrame.lookAt(destPos, lookPos)
    end

    if dist > FLIGHT_SNAP_DIST and (config.smoothTp or dist > 120) then
        local step = math.min(dist, getSafeFlySpeed() * dt)
        local nextPos = root.Position + diff.Unit * step
        local flatLook = Vector3.new(lookPos.X, nextPos.Y, lookPos.Z)
        if (flatLook - nextPos).Magnitude > 0.1 then
            root.CFrame = CFrame.lookAt(nextPos, flatLook)
        else
            root.CFrame = CFrame.new(nextPos)
        end
    elseif dist > 1.5 and (config.smoothTp or dist > FLIGHT_SNAP_DIST) then
        local step = math.min(dist, 450 * dt)
        root.CFrame = (goal - goal.Position) + (root.Position + diff.Unit * step)
    else
        root.CFrame = goal
    end
    zeroVelocity(root)
    recordSet(root)
end

local function findPart(obj)
    if obj:IsA("BasePart") then return obj end
    return obj:FindFirstChild("Handle") or obj:FindFirstChildWhichIsA("BasePart", true)
end

local function isFruit(obj)
    if not obj.Name:find("Fruit") then return false end
    if not (obj:IsA("Tool") or obj:IsA("Model") or obj:IsA("BasePart")) then return false end
    if Svc.Players:GetPlayerFromCharacter(obj) then return false end
    if obj:FindFirstChildOfClass("Humanoid") then return false end
    return true
end

local getActiveBosses
do
    local keyCache = {}
    -- Khớp CHÍNH XÁC tên Boss (cho phép hậu tố "[...]" hoặc "(...)"). Không dùng find() nữa
    -- vì quái thường như "Swan Pirate" bị nhận nhầm thành Boss "Swan".
    local function matchKey(name)
        local c = keyCache[name]
        if c ~= nil then return c or nil end
        local found = nil
        if BossQuests[name] then
            found = name
        else
            for key in pairs(BossQuests) do
                local p = name:sub(1, #key + 2)
                if p == key .. " [" or p == key .. " (" then
                    found = key
                    break
                end
            end
        end
        keyCache[name] = found or false
        return found
    end

    getActiveBosses = function()
        local list, seen = {}, {}
        local function checkFolder(folder, inWorkspace)
            if not folder then return end
            for _, m in ipairs(folder:GetChildren()) do
                if not seen[m.Name] then
                    local key = matchKey(m.Name)
                    local info = key and BossQuests[key]
                    -- chỉ nhận Boss thuộc Sea đang đứng
                    if info and (not info.sea or info.sea == CurrentSea) then
                        local hum = m:FindFirstChildOfClass("Humanoid")
                        local hrp = m:FindFirstChild("HumanoidRootPart")
                        if hum and hrp and hum.Health > 0 then
                            seen[m.Name] = true
                            list[#list + 1] = {
                                obj = m,
                                part = hrp,
                                name = m.Name,
                                bossKey = key,
                                inWorkspace = inWorkspace,
                                fallbackPos = info.pos or hrp.Position,
                            }
                        end
                    end
                end
            end
        end
        checkFolder(workspace:FindFirstChild("Enemies"), true)
        checkFolder(Svc.ReplicatedStorage, false)
        return list
    end
end

    if locs then
        for _, p in ipairs(locs:GetChildren()) do
            if p.Name:lower():find(keyword, 1, true) then return p end
        end
    end
    return nil
end

local function normalizeName(str)
    str = tostring(str or ""):lower():gsub("[^a-z]", "")
    str = str:gsub("ies$", "y")
    str = str:gsub("ves$", "f")
    str = str:gsub("men$", "man")
    str = str:gsub("s$", "")
    return str
end

local function isQuestMatching(qTitle, targetMob)
    if not qTitle or qTitle == "" or not targetMob then return false end
    if qTitle:find(targetMob, 1, true) then return true end
    local normTitle = normalizeName(qTitle)
    local normMob = normalizeName(targetMob)
    return normMob ~= "" and normTitle:find(normMob, 1, true) ~= nil
end

local function getBestQuestForLevel()
    CurrentSea = detectSea()
    local lv = playerNumber("Level") or 1
    local list = QuestData[CurrentSea] or QuestData[1]
    local chosen = list[1]
    for _, q in ipairs(list) do
        if lv >= q.lv then
            chosen = q
        else
            break
        end
    end
    return chosen
end

local function isGuiActuallyVisible(obj)
    local cur = obj
    while cur and cur ~= LP.PlayerGui do
        if cur:IsA("GuiObject") and not cur.Visible then
            return false
        elseif cur:IsA("ScreenGui") and not cur.Enabled then
            return false
        end
        cur = cur.Parent
    end
    return true
end

function QuestEngine.scanGui()
    local ok, active, allText, curKills, reqKills = pcall(function()
        local pGui = LP:FindFirstChildOfClass("PlayerGui")
        if not pGui then return false, "", nil, nil end

        local parts = {}
        local foundObjective = false
        local cKill, rKill = nil, nil

        local mainGui = pGui:FindFirstChild("Main")
        local questFrame = mainGui and mainGui:FindFirstChild("Quest")
        if questFrame and questFrame:IsA("GuiObject") and isGuiActuallyVisible(questFrame) then
            for _, d in ipairs(questFrame:GetDescendants()) do
                if d:IsA("TextLabel") and d.Text ~= "" and d.Visible then
                    parts[#parts + 1] = d.Text
                    local a, b = d.Text:match("(%d+)%s*/%s*(%d+)")
                    if a and b then
                        cKill, rKill = tonumber(a), tonumber(b)
                        foundObjective = true
                    elseif d.Text:find("Defeat ", 1, true) then
                        foundObjective = true
                    end
                end
            end
            if #parts > 0 and foundObjective then
                return true, table.concat(parts, " | "), cKill, rKill
            end
            parts = {}
            cKill, rKill = nil, nil
            foundObjective = false
        end

        for _, d in ipairs(pGui:GetDescendants()) do
            if d:IsA("TextLabel") and d.Text ~= "" and not d:IsDescendantOf(gui) then
                if isGuiActuallyVisible(d) then
                    local txt = d.Text
                    local inQuestFrame = false
                    local anc = d.Parent
                    while anc and anc ~= pGui do
                        if anc.Name:lower():find("quest", 1, true) then
                            inQuestFrame = true
                            break
                        end
                        anc = anc.Parent
                    end
                    local a, b = txt:match("(%d+)%s*/%s*(%d+)")
                    if inQuestFrame or txt:find("Defeat ", 1, true) then
                        parts[#parts + 1] = txt
                        if a and b then
                            cKill, rKill = tonumber(a), tonumber(b)
                            foundObjective = true
                        elseif txt:find("Defeat ", 1, true) then
                            foundObjective = true
                        end
                    end
                end
            end
        end

        if #parts > 0 and foundObjective then
            return true, table.concat(parts, " | "), cKill, rKill
        end
        return false, "", nil, nil
    end)
    if ok then
        return {
            active = active,
            text = allText,
            current = curKills,
            total = reqKills,
        }
    end
    return { active = false, text = "", current = nil, total = nil }
end

function QuestEngine.step(root)
    local q = getBestQuestForLevel()
    if not q then return nil, false end

    if QuestEngine.activeQuest and (QuestEngine.activeQuest.qName ~= q.qName or QuestEngine.activeQuest.qNum ~= q.qNum) then
        if QuestEngine.sawGui then
            invokeCommF("AbandonQuest")
        end
        QuestEngine.reset()
    end

    local now = os.clock()
    local guiInfo = QuestEngine.scanGui()

    if QuestEngine.phase == "FARMING" then
        if guiInfo.active then
            QuestEngine.sawGui = true
            if guiInfo.current and guiInfo.total then
                if guiInfo.current > QuestEngine.lastKillCount then
                    QuestEngine.lastKillCount = guiInfo.current
                    QuestEngine.lastProgressAt = now
                end
                if guiInfo.current >= guiInfo.total then
                    QuestEngine.reset()
                end
            end
        else
            if QuestEngine.sawGui and (now - QuestEngine.acceptedAt) >= 3.5 then
                QuestEngine.reset()
            elseif not QuestEngine.sawGui and (now - QuestEngine.acceptedAt) >= 90 then
                QuestEngine.reset()
            end
        end

        if QuestEngine.phase == "FARMING" then
            return q, true
        end
    end

    if guiInfo.active then
        local matching = isQuestMatching(guiInfo.text, q.mob)
        local finished = guiInfo.current and guiInfo.total and guiInfo.current >= guiInfo.total
        if matching and not finished then
            QuestEngine.phase = "FARMING"
            QuestEngine.activeQuest = q
            QuestEngine.sawGui = true
            QuestEngine.acceptedAt = now
            QuestEngine.lastProgressAt = now
            QuestEngine.lastKillCount = guiInfo.current or 0
            QuestEngine.abandonTries = 0
            return q, true
        elseif matching and finished then
        elseif QuestEngine.abandonTries < 3 then
            if now - QuestEngine.lastRemoteTry >= 1.0 then
                QuestEngine.lastRemoteTry = now
                QuestEngine.abandonTries = QuestEngine.abandonTries + 1
                invokeCommF("AbandonQuest")
            end
            return q, false
        end
    end

    if now - QuestEngine.lastRemoteTry >= 1.2 then
        QuestEngine.lastRemoteTry = now
        invokeCommF("StartQuest", q.qName, q.qNum)
        for _ = 1, 5 do
            local afterGui = QuestEngine.scanGui()
            local afterDone = afterGui.current and afterGui.total and afterGui.current >= afterGui.total
            if afterGui.active and isQuestMatching(afterGui.text, q.mob) and not afterDone then
                QuestEngine.phase = "FARMING"
                QuestEngine.activeQuest = q
                QuestEngine.sawGui = true
                QuestEngine.acceptedAt = os.clock()
                QuestEngine.lastProgressAt = os.clock()
                QuestEngine.lastKillCount = afterGui.current or 0
                QuestEngine.abandonTries = 0
                break
            end
            task.wait(0.15)
        end
        return q, QuestEngine.phase == "FARMING"
    end

    return q, false
end

local function findMobByName(mobName, root, maxRange)
    local folder = workspace:FindFirstChild("Enemies")
    if not folder then return nil end
    local best, bestD = nil, maxRange or math.huge
    for _, m in ipairs(folder:GetChildren()) do
        if isAlive(m) then
            local match = (not mobName) or (m.Name == mobName) or (m.Name:find(mobName, 1, true) ~= nil)
            if match then
                local d = (m.HumanoidRootPart.Position - root.Position).Magnitude
                if d < bestD then

                end
            end
        end
    end
    return best

        if isAlive(m) then
            for _, targetName in ipairs(nameList) do
                if m.Name == targetName or m.Name:find(targetName, 1, true) then
                    local d = (m.HumanoidRootPart.Position - root.Position).Magnitude
                    if d < bestD then
                        best, bestD = m, d
                    end
                    break
                end
            end
        end
    end
    return best
end

local function findSeaTarget(keywords, root, maxRange)
    local best, bestD = nil, maxRange or math.huge
    for _, containerName in ipairs({ "Enemies", "SeaBeasts" }) do
        local folder = workspace:FindFirstChild(containerName)
        if folder then
            for _, obj in ipairs(folder:GetChildren()) do
                local matched = (not keywords)
                if not matched then
                    for _, kw in ipairs(keywords) do
                        if obj.Name:lower():find(kw:lower(), 1, true) then
                            matched = true
                            break
                        end
                    end
                end
                if matched then
                    local hum = obj:FindFirstChildOfClass("Humanoid")
                    local part = obj:FindFirstChild("HumanoidRootPart") or findPart(obj)
                    local alive = part and (not hum or hum.Health > 0)
                    if alive then
                        local d = (part.Position - root.Position).Magnitude
                        if d < bestD then
                            best, bestD = { obj = obj, part = part, name = obj.Name, isMob = hum ~= nil }, d
                        end
                    end
                end
            end
        end
    end
    return best
end

local function applyBringMob(primaryMob, allInRange)
    if not config.bringMob or not isAlive(primaryMob) then return end
    local folder = workspace:FindFirstChild("Enemies")
    if not folder then return end
    local anchorCF = primaryMob.HumanoidRootPart.CFrame
    local anchorPos = anchorCF.Position
    if anchorPos.Y < -5 then return end
    local targetName = primaryMob.Name

    pcall(function()
        if sethiddenproperty then
            sethiddenproperty(LP, "SimulationRadius", math.huge)
        end
    end)

    for _, m in ipairs(folder:GetChildren()) do
        if isAlive(m) and (allInRange or m.Name == targetName) then
            local hrp = m:FindFirstChildOfClass("HumanoidRootPart") or m:FindFirstChild("HumanoidRootPart")
            local hum = m:FindFirstChildOfClass("Humanoid")
            if hrp and hum and (hrp.Position - anchorPos).Magnitude <= config.bringRange then
                pcall(function()
                    hrp.CFrame = anchorCF
                    hrp.AssemblyLinearVelocity = Vector3.zero
                    hrp.AssemblyAngularVelocity = Vector3.zero
                    hrp.CanCollide = false
                    hrp.Size = Vector3.new(60, 60, 60)
                    hrp.Transparency = 1
                    hum.WalkSpeed = 0
                    hum.JumpPower = 0
                    if hum:FindFirstChild("Animator") then
                        hum.Animator:Destroy()
                    end
                end)
            end
        end
    end
end

local function findNearestEnemyPlayer(root, maxRange, trialArenaOnly)
    local bestChar, bestHrp, bestName, bestD = nil, nil, nil, maxRange or math.huge
    for _, p in ipairs(Svc.Players:GetPlayers()) do
        if p ~= LP and p.Character then
            local hum = p.Character:FindFirstChildOfClass("Humanoid")
            local hrp = p.Character:FindFirstChild("HumanoidRootPart")
            if hum and hrp and hum.Health > 0 then
                local inArena = (not trialArenaOnly) or (hrp.Position.Y > 13500 and (hrp.Position - TempleOfTimePos).Magnitude <= 1500)
                if inArena then
                    local d = (hrp.Position - root.Position).Magnitude
                    if d < bestD then
                        bestChar, bestHrp, bestName, bestD = p.Character, hrp, p.DisplayName, d
                    end
                end
            end
        end
    end
    return bestChar, bestHrp, bestName
end

                    end

    return true
end

-- Kích hoạt rương / cửa / nút gần vị trí (chỉ vật thể có tên liên quan)
function SW.touchNear(root, pos, radius)
    local now = os.clock()
    if now - SW.lastTouch < 1.5 then return end
    SW.lastTouch = now
    local ok, parts = pcall(workspace.GetPartBoundsInRadius, workspace, pos, radius)
    if not ok or type(parts) ~= "table" then return end
    local used = 0
    for _, p in ipairs(parts) do
        if used >= 20 then break end
        local nm = (p.Name .. " " .. (p.Parent and p.Parent.Name or "")):lower()
        if nm:find("chest", 1, true) or nm:find("door", 1, true) or nm:find("hidden", 1, true)
            or nm:find("rengoku", 1, true) or nm:find("key", 1, true) then
            local cd = p:FindFirstChildOfClass("ClickDetector")
            local pp = p:FindFirstChildOfClass("ProximityPrompt")
            if cd and fireclickdetector then
                used = used + 1
                pcall(fireclickdetector, cd)
            elseif pp and fireproximityprompt then
                used = used + 1
                pcall(fireproximityprompt, pp)
            elseif p:FindFirstChildOfClass("TouchTransmitter") and firetouchinterest then
                used = used + 1
                pcall(function()
                    firetouchinterest(root, p, 0)
                    firetouchinterest(root, p, 1)
                end)
            end
        end
    end
end

-- Kiếm thuộc Sea khác: tự thử chuyển Sea, không bay vào hư không

function SW.step(root, dt)
    local sw = getSwordSpec(config.selectedSword)
    if not sw then return end

    -- 0. Đã có kiếm -> xong
    if hasToolInInventory(sw.id) then
        setToggle("autoGetSword", false)
        notify("Lấy Kiếm Thành Công!", "Bạn đã sở hữu: " .. sw.id, 5, T.good, true)
        State.target = nil
        SW.id = nil
        return
    end

    -- 1. Đúng Sea chưa?
    if not SW.ensureSea(sw) then return end

    -- 2. Boss rơi kiếm
    local bossNames = nil
    if sw.id == "Spikey Trident" then
        bossNames = { "Dough King", "Cake Prince" }
    elseif sw.boss then
        bossNames = { sw.boss }
    end
    if bossNames then
        local mob = findMobFromList(bossNames, root, math.huge)
        if isAlive(mob) then
            SW.setMode(sw.id, "boss")
            SW.fight(root, dt, mob, ("Lấy Kiếm (%s): Diệt %s"):format(sw.id, mob.Name), false)
            return
        end
        if SW.flyToSpawned(root, dt, bossNames, sw.id, ("Lấy Kiếm (%s): %s đã spawn - đang bay tới"):format(sw.id, bossNames[1])) then
            return
        end
    end

    -- 3. Xử lý riêng từng kiếm
    if sw.id == "Rengoku" then
        if sw.chestPos and hasToolInInventory("Hidden Key") then
            SW.setMode(sw.id, "chest")
            State.target = nil
            State.statusText = "Rengoku: Mở rương bí mật dưới cầu thang!"
            moveTowards(root, sw.chestPos, sw.chestPos, dt)

-- =====================================================
--  BOSS ENGINE (SỬA LỖI: bay qua bay lại / tìm không ra Boss)
--  - Khớp CHÍNH XÁC tên Boss (trước đây "Swan Pirate" bị nhận nhầm là Boss "Swan")
--  - Chỉ nhận Boss đúng Sea đang đứng (trước đây bay sang tọa độ Sea khác)
--  - Chọn 1 Boss rồi bám tới khi xong (trước đây bosses[1] đổi liên tục -> bay qua bay lại)
--  - Nhận nhiệm vụ Boss tối đa 1 lần / Boss (trước đây không thấy NV là quay lại NPC mãi)
--  - Boss chưa load: bay tới vị trí thật; tới nơi mà không thấy thì bỏ qua 2 phút, qua Boss khác
-- =====================================================
local BOSS = {
    selKey = nil,
    list = {}, lastScan = -1e9,
    blacklist = {},                 -- [bossKey] = thời điểm hết hạn bỏ qua
    questTried = {},                -- [bossKey] = lần cuối đã thử nhận NV
    npcStart = {},                  -- [bossKey] = lúc bắt đầu bay tới NPC
    atNpcSince = nil,
    far = { key = nil, phase = 1, arrived = nil },
    idleSince = 0, lastHop = -1e9,
    hop = nil,                      -- gán sau khi serverHop được định nghĩa
}

function BOSS.pos(b)
    local p = b.part and b.part.Position
    if p and p.Y > -50 and p.Magnitude > 1 then return p end
    return b.fallbackPos
end

function BOSS.pick(root)
    local now = os.clock()
    if now - BOSS.lastScan > 0.25 then
        BOSS.lastScan = now
        BOSS.list = getActiveBosses()
    end
    local cands = {}
    for _, b in ipairs(BOSS.list) do
        local untilT = BOSS.blacklist[b.bossKey]
        if b.obj and b.obj.Parent and (not untilT or now > untilT) then
            cands[#cands + 1] = b
        end
    end

    local function nearest(onlyLoaded)
        local best, bestD = nil, math.huge
        for _, b in ipairs(cands) do
            if not onlyLoaded or b.inWorkspace then
                local d = (BOSS.pos(b) - root.Position).Magnitude
                if d < bestD then best, bestD = b, d end
            end
        end
        return best
    end

    -- Giữ nguyên mục tiêu đang chọn (không đổi qua lại), trừ khi có Boss đã load sẵn mà mục tiêu hiện tại thì chưa
    if BOSS.selKey then
        for _, b in ipairs(cands) do
            if b.bossKey == BOSS.selKey then
                if not b.inWorkspace then
                    local loaded = nearest(true)
                    if loaded then
                        BOSS.selKey = loaded.bossKey
                        return loaded, #cands
                    end
                end
                return b, #cands
            end
        end
        BOSS.selKey = nil
    end

    local best = nearest(true) or nearest(false)
    if best then
        BOSS.selKey = best.bossKey
        BOSS.far.key = nil
    end
    return best, #cands
end

function BOSS.step(root, dt)
    local now = os.clock()
    local b, n = BOSS.pick(root)

    -- Không có Boss nào
    if not b then
        State.target = nil
        BOSS.atNpcSince = nil
        if BOSS.idleSince == 0 then BOSS.idleSince = now end
        local waited = math.floor(now - BOSS.idleSince)
        State.statusText = ("Chưa có Boss nào ở Sea %d (đã chờ %ds)"):format(CurrentSea, waited)
        if config.bossAutoHop and BOSS.hop and waited > 45 and now - BOSS.lastHop > 120 then
            BOSS.lastHop = now
            notify("Săn Boss", "Không có Boss - đang đổi server...", 5, T.accent, true)
            BOSS.hop()
        end
        return
    end
    BOSS.idleSince = 0

    local key = b.bossKey
    local bq = BossQuests[key]

    -- 1. Nhận nhiệm vụ Boss: tối đa 1 lần mỗi Boss, KHÔNG quay lại NPC nữa dù GUI có nhận ra hay không
    if config.autoBossQuest and bq and bq.qName and bq.npc then
        local tried = BOSS.questTried[key]
        if tried and now - tried < 240 then
            BOSS.atNpcSince = nil
            BOSS.npcStart[key] = nil
        elseif hasBossOrSpecialQuest(key) then
            BOSS.questTried[key] = now
        else
            local d = (root.Position - bq.npc).Magnitude
            local giveUp = false
            if d > 35 then
                BOSS.atNpcSince = nil
                BOSS.npcStart[key] = BOSS.npcStart[key] or now
                if now - BOSS.npcStart[key] > 60 then giveUp = true end
            else
                BOSS.atNpcSince = BOSS.atNpcSince or now
                if now - BOSS.atNpcSince > 4 then giveUp = true end -- vòng lặp remote đã gửi StartQuest
            end
            if giveUp then
                BOSS.questTried[key] = now
                BOSS.atNpcSince = nil
                BOSS.npcStart[key] = nil
            else
                State.target = nil
                State.statusText = "Nhận NV Boss: " .. key
                moveTowards(root, bq.npc + Vector3.new(0, 5, 0), bq.npc, dt)
                return
            end
        end
    end

    -- 2. Boss đã load: đánh
    if b.inWorkspace and isAlive(b.obj) then
        BOSS.far.key = nil
        State.target = b.obj
        State.statusText = ("Đánh Boss: %s (%d Boss)"):format(b.name, n)
        local p = b.part.Position
        moveTowards(root, p + Vector3.new(0, config.farmHeight, config.farmDist), p, dt)
        return
    end

    -- 3. Boss chưa load (nằm trong ReplicatedStorage): bay tới vị trí thật, không thấy thì thử tọa độ cố định rồi bỏ qua
    State.target = nil
    local f = BOSS.far
    if f.key ~= key then
        f.key = key
        f.phase = 1
        f.arrived = nil
    end
    local target = (f.phase == 1) and BOSS.pos(b) or b.fallbackPos
    local dist = (root.Position - target).Magnitude
    if dist < 200 then
        f.arrived = f.arrived or now
        if now - f.arrived > 12 then
            if f.phase == 1 and (b.fallbackPos - target).Magnitude > 300 then
                f.phase = 2
                f.arrived = nil
            else
                BOSS.blacklist[key] = now + 120
                BOSS.selKey = nil
                f.key = nil
                notify("Săn Boss", key .. " không xuất hiện - bỏ qua 2 phút", 4, T.bad, true)
                return
            end
        end
    else
        f.arrived = nil
    end
    State.statusText = ("Bay tới Boss: %s (%d studs)"):format(key, math.floor(dist))
    moveTowards(root, target + Vector3.new(0, 25, 0), target, dt)
end

                                if desc:IsA("ClickDetector") then pcall(fireclickdetector, desc) end
                            end
                        end
                    end
                end
            end
            if config.autoAwaken then invokeCommF("Awakener", "Awaken") end
            if config.autoMirageGear then
                local moonDir = Svc.Lighting:GetMoonDirection()
                local cam = workspace.CurrentCamera
                if cam and moonDir.Magnitude > 0 then
                    cam.CFrame = CFrame.lookAt(cam.CFrame.Position, cam.CFrame.Position + moonDir * 100)
                end
                local commE = getRemote("CommE")
                if commE then pcall(commE.FireServer, commE, "ActivateAbility") end
            end
            if config.autoUpgradeV4 then invokeCommF("UpgradeRace", "Buy") end
        end
        end)
        task.wait(0.65)
    end
end)

local function checkHasFruitOver1M()
    local char = LP.Character
    for _, container in ipairs({ LP.Backpack, char }) do
        if container then
            for _, item in ipairs(container:GetChildren()) do
                if item:IsA("Tool") and item.Name:find("Fruit") then
                    local res = invokeCommF("TalkTrevor", "1")
                    local res2 = invokeCommF("TalkTrevor", "2")
                    if res == 0 or res2 == 0 then
                        return true
                    end
                end
            end
        end
    end
    return false
end

local function updateSea3Progression()
    CurrentSea = detectSea()
    if CurrentSea >= 3 then
        Sea3State.task = "DONE"
        Sea3State.stepDesc = "Đã ở Sea 3!"
        return
    end

    local lv = playerNumber("Level") or 1

    if CurrentSea == 1 then
        if lv < 700 then
            Sea3State.task = "LEVEL"
            Sea3State.stepDesc = string.format("Sea 1: Cày Level (%d/700)", lv)
            return
        end

        local prog = invokeCommF("DressrosaQuestProgress")
        if prog == 2 then
            Sea3State.task = "TRAVEL_SEA2"
            Sea3State.stepDesc = "Đủ điều kiện! Đang dịch chuyển sang Sea 2..."
            invokeCommF("TravelDressrosa")
        else
            Sea3State.task = "SEA2_UNLOCK"
            if prog == 0 then
                Sea3State.stepDesc = "Lấy chìa khóa từ Military Detective (Prison)"
            else
                Sea3State.stepDesc = "Diệt Ice Admiral tại Hang Băng (Frozen Village)"
            end
        end
        return
    end

    if CurrentSea == 2 then
        if lv < 850 then
            Sea3State.task = "LEVEL"
            Sea3State.stepDesc = string.format("Sea 2: Cày lên Lv 850 mở Bartilo (%d/850)", lv)
            return
        end

        local bartiloProg = invokeCommF("BartiloQuestProgress")
        if bartiloProg ~= 3 then
            Sea3State.task = "BARTILO"
            if bartiloProg == 0 then
                Sea3State.stepDesc = "Bartilo (1/3): Diệt 50 Swan Pirate"
            elseif bartiloProg == 1 then
                Sea3State.stepDesc = "Bartilo (2/3): Diệt Boss Jeremy"
            else
                Sea3State.stepDesc = "Bartilo (3/3): Giải mật mã hầm Colosseum"
            end
            return
        end

        if lv < 1500 then
            Sea3State.task = "LEVEL"
            Sea3State.stepDesc = string.format("Sea 2: Cày Level lên Sea 3 (%d/1500)", lv)
            return
        end

        local zouProg = invokeCommF("ZouQuestProgress")
        if zouProg == 3 then
            Sea3State.task = "TRAVEL_SEA3"
            Sea3State.stepDesc = "Hoàn thành! Đang sang Sea 3..."
            invokeCommF("TravelZou")
        elseif zouProg == 2 then
            Sea3State.task = "RIP_INDRA"
            Sea3State.stepDesc = "Đánh Boss Rip_Indra (Mở cổng Sea 3)"
        else
            Sea3State.task = "DON_SWAN"
            Sea3State.stepDesc = "Mở phòng & Diệt Boss Don Swan"
        end
    end
end

            "Saber Expert", "Thunder God", "Darkbeard", "rip_indra True Form",

    if config.autoGetMelee then
        if config.selectedMelee == "Sharkman Karate" then
            local hasKey = LP.Backpack:FindFirstChild("Water Key") or char:FindFirstChild("Water Key")
            if not hasKey then
                local tide = findMobByName("Tide Keeper", root, math.huge)
                if isAlive(tide) then
                    State.target = tide
                    State.statusText = "Lấy Võ: Đánh Tide Keeper lấy Water Key"
                    local pos = tide.HumanoidRootPart.Position
                    moveTowards(root, pos + Vector3.new(0, config.farmHeight, config.farmDist), pos, dt)
                else
                    local tPos = Vector3.new(-3711, 90, -11469)
                    State.statusText = "Chờ Tide Keeper hồi sinh..."

                end
                return

                moveTowards(root, MansionPos, MansionPos, dt)
                return
            end

            return
        end

        if now - Sea3State.lastCheck >= 2.0 then

        end

        if Sea3State.task == "SEA2_UNLOCK" then
            State.statusText = Sea3State.stepDesc
            local prog = invokeCommF("DressrosaQuestProgress")
            if prog == 0 then
                local detectivePos = Vector3.new(4849, 5, 719)
                moveTowards(root, detectivePos, detectivePos, dt)
                if (root.Position - detectivePos).Magnitude <= 15 then
                    invokeCommF("DressrosaQuestProgress", "Detective")
                end
            else
                local iceBoss = findMobByName("Ice Admiral", root, math.huge)
                if isAlive(iceBoss) then
                    State.target = iceBoss
                    local pos = iceBoss.HumanoidRootPart.Position
                    moveTowards(root, pos + Vector3.new(0, config.farmHeight, config.farmDist), pos, dt)
                else
                    local cavePos = Vector3.new(1347, 37, -1325)
                    moveTowards(root, cavePos, cavePos, dt)
                end
            end
            return
        end

        if Sea3State.task == "BARTILO" then
            State.statusText = Sea3State.stepDesc
            local bProg = invokeCommF("BartiloQuestProgress")
            if bProg == 0 then
                if not hasBossOrSpecialQuest("Swan Pirate") then
                    local bartiloPos = Vector3.new(-456, 73, 300)
                    moveTowards(root, bartiloPos, bartiloPos, dt)
                    if (root.Position - bartiloPos).Magnitude <= 15 then
                        invokeCommF("StartQuest", "BartiloQuest", 1)
                    end
                else
                    local swan = findMobByName("Swan Pirate", root, math.huge)
                    if isAlive(swan) then
                        State.target = swan
                        applyBringMob(swan, false)
                        local pos = swan.HumanoidRootPart.Position
                        moveTowards(root, pos + Vector3.new(0, config.farmHeight, config.farmDist), pos, dt)
                    else
                        local swanPos = Vector3.new(1068, 137, 1322)
                        moveTowards(root, swanPos, swanPos, dt)
                    end
                end
            elseif bProg == 1 then
                local jeremy = findMobByName("Jeremy", root, math.huge)
                if isAlive(jeremy) then
                    State.target = jeremy
                    local pos = jeremy.HumanoidRootPart.Position
                    moveTowards(root, pos + Vector3.new(0, config.farmHeight, config.farmDist), pos, dt)
                else
                    local jeremyPos = Vector3.new(2316, 449, 787)
                    moveTowards(root, jeremyPos, jeremyPos, dt)
                end
            elseif bProg == 2 then
                local puzzlePos = Vector3.new(-1837, 10, 1682)
                moveTowards(root, puzzlePos, puzzlePos, dt)

                    invokeCommF("BartiloQuestProgress", "DidPlates")
                end
            end
            return
        end

        if Sea3State.task == "DON_SWAN" then
            State.statusText = Sea3State.stepDesc
            local trevorPos = Vector3.new(-303, 331, 597)
            local swanBoss = findMobByName("Don Swan", root, math.huge)
            if isAlive(swanBoss) then
                State.target = swanBoss
                local pos = swanBoss.HumanoidRootPart.Position
                moveTowards(root, pos + Vector3.new(0, config.farmHeight, config.farmDist), pos, dt)
            else

            end

        end

        if Sea3State.task == "RIP_INDRA" then
            State.statusText = Sea3State.stepDesc

            local indra = findMobByName("rip_indra", root, math.huge)
            if isAlive(indra) then
                State.target = indra
                local pos = indra.HumanoidRootPart.Position
                moveTowards(root, pos + Vector3.new(0, config.farmHeight, config.farmDist), pos, dt)

                moveTowards(root, redHeadPos, redHeadPos, dt)
                if (root.Position - redHeadPos).Magnitude <= 15 then
                    invokeCommF("ZouQuestProgress", "Check")
                    invokeCommF("TravelZou")
                end
            end
            return
        end
    end

    if config.autoToSea3 or config.autoFarmLevel or config.autoMastery or config.autoFarmMeleeMastery or config.autoSuperhuman or config.autoGodhuman then

            local mName, mLv = getCurrentMeleeInfo()
            if mName ~= "None" and mLv >= config.targetMeleeMastery then

        if not q then return end

        if not readyToFarm then
            State.target = nil
            State.statusText = "Nhận NV mới: " .. q.mob
            moveTowards(root, q.npc + Vector3.new(0, 4, 0), q.npc, dt)
            return
        end

        if not isAlive(State.target) or not State.target.Name:find(q.mob, 1, true) then
            if now - State.lastScan >= 0.15 then
                State.lastScan = now
                State.target = findMobByName(q.mob, root, math.huge)
            end
        end

        if isAlive(State.target) then
            if config.autoFarmMeleeMastery or config.autoSuperhuman or config.autoGodhuman then
                local mName, mLv = getCurrentMeleeInfo()
                State.statusText = string.format("Cày Võ %s (Lv %d)", mName, mLv)
            else
                State.statusText = (config.autoMastery and "Mastery: " or "Đang cày: ") .. State.target.Name
            end
            applyBringMob(State.target, false)
            local mobPos = State.target.HumanoidRootPart.Position
            moveTowards(root, mobPos + Vector3.new(0, config.farmHeight, config.farmDist), mobPos, dt)
        else
            local spawns = getMobSpawnPoints(q.mob, q.mobPos)
            if now - State.lastSpawnSwitch >= 2.5 then
                State.lastSpawnSwitch = now
                State.spawnIndex = (State.spawnIndex % #spawns) + 1
            end
            local waitPos = spawns[State.spawnIndex] or q.mobPos
            State.statusText = "Chờ quái hồi: " .. q.mob
            moveTowards(root, waitPos + Vector3.new(0, math.max(config.farmHeight, 18), 0), waitPos, dt)
        end
        return
    end

    -- 11. SĂN QUÁI THEO TÊN HOẶC CÀY QUÁI GẦN NHẤT
    local huntName = (config.huntOnly and config.huntName) or nil
    if not isAlive(State.target) or (huntName and State.target.Name ~= huntName) then
        if now - State.lastScan >= 0.2 then
            State.lastScan = now
            State.target = findMobByName(huntName, root, huntName and math.huge or config.farmRange)
        end
    end

    if isAlive(State.target) then
        State.statusText = "Đánh: " .. State.target.Name
        applyBringMob(State.target, false)
        local mobPos = State.target.HumanoidRootPart.Position
        moveTowards(root, mobPos + Vector3.new(0, config.farmHeight, config.farmDist), mobPos, dt)
    else
        State.statusText = huntName and ("Chờ: " .. huntName) or "Không có quái gần đây"
    end
    end)
end)

                if config.autoFarmMeleeMastery or config.autoSuperhuman or config.autoGodhuman then

        if config.autoFruit and os.clock() >= State.busyUntil then
            local _, hum, root = getChar()
            if root and hum and hum.Health > 0 then
                local best = nearestFruit(root)
                if best then
                    local dur = teleportTo(best.part.CFrame)
                    State.busyUntil = os.clock() + dur + 1.5
                    if firetouchinterest then
                        pcall(function() firetouchinterest(root, best.part, 0) firetouchinterest(root, best.part, 1) end)
                    end
                    notify("Nhặt trái", best.name, 2, T.accent)
                end
            end
        end

        if config.fruitAlert then
            local _, _, root = getChar()
            local current = {}
            for _, f in ipairs(getFruits()) do
                current[f.obj] = true
                if not State.knownFruits[f.obj] then
                    State.knownFruits[f.obj] = true
                    local detail = f.name
                    if root then
                        detail = detail .. " - cách " .. tostring(math.floor((f.part.Position - root.Position).Magnitude)) .. " đv"
                    end
                    notify("Có trái ác quỷ!", detail, 6, T.accent, true)
                end
            end
            for obj in pairs(State.knownFruits) do
                if not current[obj] then State.knownFruits[obj] = nil end
            end
        else
            table.clear(State.knownFruits)
        end
        task.wait(1.5)
    end
end)

-- Tự cộng điểm chỉ số
task.spawn(function()
    local lastPts, stuck = nil, 0
    while State.running do
        if config.autoStats then
            local pts = playerNumber("Points")
            if not pts then
                if not State.statWarned then
                    State.statWarned = true
                    notify("Cộng điểm", "Không đọc được số điểm còn lại", 5, T.bad, true)
                end
            elseif pts > 0 then
                if pts == lastPts then stuck = stuck + 1 else stuck = 0 end
                lastPts = pts
                if stuck >= 3 and not State.statWarned then
                    State.statWarned = true
                    notify("Cộng điểm", "Điểm không giảm sau 3 lần thử", 5, T.bad, true)
                end
                invokeCommF("AddPoint", config.statType, math.min(pts, config.statStep))
            end
        else
            lastPts, stuck = nil, 0
        end
        task.wait(1.5)
    end
end)

local function boostFPS()
    pcall(function() settings().Rendering.QualityLevel = Enum.QualityLevel.Level01 end)
    for _, v in ipairs(workspace:GetDescendants()) do
        pcall(function()
            if v:IsA("BasePart") then
                v.Material = Enum.Material.SmoothPlastic
                v.Reflectance = 0
                v.CastShadow = false
            elseif v:IsA("Decal") or v:IsA("Texture") then
                v.Transparency = 1
            elseif v:IsA("ParticleEmitter") or v:IsA("Trail") or v:IsA("Smoke") or v:IsA("Fire") or v:IsA("Sparkles") then
                v.Enabled = false
            end
        end)
    end
    notify("Giảm Lag (FPS Boost)", "Đã tối ưu đồ họa mượt hơn!", 3, T.good, true)
end

local function serverHop()
    notify("Đổi server", "Đang tìm server còn chỗ...", 3)
    task.spawn(function()
        local ok, found = pcall(function()
            local url = ("https://games.roblox.com/v1/games/%d/servers/Public?sortOrder=Asc&limit=100"):format(game.PlaceId)
            local data = Svc.HttpService:JSONDecode(game:HttpGet(url))
            for _, s in ipairs(data.data or {}) do
                if s.id ~= game.JobId and s.playing < s.maxPlayers - 1 then
                    Svc.TeleportService:TeleportToPlaceInstance(game.PlaceId, s.id, LP)
                    return true
                end
            end
            return false
        end)
        if not ok or not found then
            notify("Đổi server", "Không tìm được server phù hợp", 4, T.bad)
        end
    end)
end

end

local function stopExclusiveFarm(keepKey)
    cancelTeleport()
    QuestEngine.reset()
    for _, k in ipairs(exclusiveKeys) do
        if k ~= keepKey and config[k] then
            setToggle(k, false)
        end
    end
    if keepKey and config.fly then
        setToggle("fly", false)
        stopFly()
    end
    State.target = nil
    State.targetPlayer = nil
    State.busyUntil = 0
    State.lastScan = 0
    State.lastSpawnSwitch = 0
    State.spawnIndex = 1
    if not anyAutoActive() then
        stopFarmBody()
        if not config.noclip and not State.isTeleporting then restoreNoclip() end
    end
end

local function getEnemyNames()
    local list, seen = {}, {}
    local folder = workspace:FindFirstChild("Enemies")
    if folder then
        for _, m in ipairs(folder:GetChildren()) do
            if isAlive(m) and not seen[m.Name] then
                seen[m.Name] = true
                list[#list + 1] = { label = m.Name, value = m.Name }
            end
        end
    end
    table.sort(list, function(a, b) return a.label < b.label end)
    return list
end

local function commas(v)
    return (tostring(v):reverse():gsub("(%d%d%d)", "%1,"):reverse():gsub("^,", ""))
end

local function playerStat(name)
    local v = playerNumber(name)
    return v and commas(v) or "—"
end

local frames, lastT = 0, os.clock()
connect(Svc.RunService.RenderStepped, function() frames = frames + 1 end)
task.spawn(function()
    while State.running do
        local now = os.clock()
        local fps = math.floor(frames / math.max(now - lastT, 0.001))
        frames, lastT = 0, now

        local okP, ping = pcall(function()
            return Svc.Stats.Network.ServerStatsItem["Data Ping"]:GetValue()
        end)
        local pingStr = okP and (tostring(math.floor(ping)) .. " ms") or "—"
        infoFPS.Text = tostring(fps) .. " FPS • " .. pingStr

        local t = workspace.DistributedGameTime
        infoPlayers.Text = string.format("%d/%d • %02d:%02d",
            #Svc.Players:GetPlayers(), Svc.Players.MaxPlayers, math.floor(t / 60), math.floor(t % 60))

        local mName, mLv = getCurrentMeleeInfo()
        infoCurMelee.Text = string.format("%s (Lv %d)", mName, mLv)

        infoSea.Text = "Sea " .. tostring(CurrentSea)
        infoRace.Text = getPlayerRace()
        infoLevel.Text = playerStat("Level")
        infoBeli.Text = playerStat("Beli")
        infoFrag.Text = playerStat("Fragments")
        infoPoints.Text = playerStat("Points")
        infoStatus.Text = State.statusText

        local bestQ = getBestQuestForLevel()
        infoQuest.Text = bestQ and (bestQ.mob .. " (Lv " .. tostring(bestQ.lv) .. "+)") or "—"
        task.wait(0.5)
    end
end)

State.unload = function()
    State.running = false
    State.target = nil
    State.targetPlayer = nil
    QuestEngine.reset()
    cancelTeleport()
    stopFly()
    stopFarmBody()
    pcall(function() Svc.RunService:Set3dRenderingEnabled(true) end)
    pcall(function()
        local noti = LP.PlayerGui:FindFirstChild("Notifications")
        if noti then noti.Enabled = true end
    end)

    local _, hum = getChar()
    if hum and config.speedBoost then
        hum.WalkSpeed = State.origWalk
        hum.JumpPower = State.origJump
    end
    restoreNoclip()
    applyFullbright(false)
    for _, g in ipairs(espGroups) do clearGroup(g) end
    if State.currentTween then State.currentTween:Cancel() end

    for _, c in ipairs(State.connections) do c:Disconnect() end
    table.clear(State.connections)
    gui:Destroy()
    if env.BF_Unload == State.unload then env.BF_Unload = nil end
end
env.BF_Unload = State.unload
